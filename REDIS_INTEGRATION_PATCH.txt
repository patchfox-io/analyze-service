            }
        } else {

3. ADD before the return statement (around line 1207):

        // ===== REDIS CACHE INTEGRATION: SAVE TO REDIS =====
        log.info("Saving caches to Redis for next page");
        cacheService.saveCaches(
            datasetName,
            pageIndex,
            historicalPackagePurlsByDatasourcePurl,
            historicalFindingsByDatasourcePurl,
            historicalDatasetEditsByCommitDateAsc
        );
        // ===== END REDIS CACHE INTEGRATION =====

        return ApiResponse.builder()
                          .txid(txid)
                          .requestReceivedAt(requestReceivedAt)
                          .code(statusCode)
                          .data(data)
                          .build();


4. Add to pom.xml dependencies:

        <dependency>
            <groupId>org.springframework.boot</groupId>
            <artifactId>spring-boot-starter-data-redis</artifactId>
        </dependency>


5. Add to application.properties:

spring.data.redis.host=localhost
spring.data.redis.port=6379
spring.data.redis.timeout=60000


NOTES:
- Cache key format: "tabulate:{packages|findings|edits}:{datasetName}:{pageIndex}"
- Redis will serialize the ConcurrentHashMaps using Jackson JSON
- On cache hit, skips ~2-5 minutes of database queries
- On cache miss, falls back to existing hydration logic
- Caches are saved at end of each page for next page to use
